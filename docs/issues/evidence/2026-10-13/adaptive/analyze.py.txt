from pathlib import Path
from collections import Counter
import hashlib
import json
import statistics
import matplotlib
matplotlib.use('Agg')
import matplotlib.pyplot as plt

root = Path(__file__).resolve().parent
base = json.loads((root/'results.json').read_text())
parking_root = root.parent/'turbotest-adaptive-parking-20261013'
parking = json.loads((parking_root/'results.json').read_text())

def durations(rows, start_kind, end_kind):
    values = []
    for row in rows:
        starts = {}
        for event in row['events']:
            if event['Kind'] == start_kind:
                starts[event['Slot']] = event['TimeMs']
            elif event['Kind'] == end_kind:
                values.append(event['TimeMs']-starts.pop(event['Slot']))
        assert not starts, (row['name'],start_kind,starts)
    return values

for folder, result, expected in [(root,base,{130:900,1300:450,13000:150}), (parking_root,parking,{130:1296,1300:648,13000:216})]:
    manifest = json.loads((folder/'run.json').read_text())
    assert manifest.get('exitCode',manifest.get('exit')) == 0
    assert Counter(s['InsertedRows'] for s in result['samples']) == expected
    assert all(hashlib.sha256((folder/name).read_bytes()).hexdigest() == digest for name,digest in manifest['sourceSha256'].items())
    for row in result['results']:
        stats = row['stats']
        assert stats['MaxReserved'] <= 8 and stats['MaxPreparing'] <= 1 and stats['MaxActiveResets'] <= 2
        assert stats['LifecycleErrors'] == 0 and row['Errors'] == 0
        assert all(point['State']['Reserved'] <= 8 for point in row['timeline'])

park_rows = [r for r in parking['results'] if r['name'].startswith('park-')]
drop_rows = [r for r in parking['results'] if r['name'].startswith('drop-')]
park_ms = durations(park_rows,'park-start','park-complete')
rewarm_ms = durations(park_rows,'rewarm-start','rewarm-ready')
drop_ms = durations(drop_rows,'retire-start','retire-complete')
for row in park_rows:
    idle = row['idleSnapshot']['pool']
    assert idle['Capacity'] == 2 and idle['Parked'] == 6 and idle['Reserved'] == 8
    assert row['stats']['Created'] == 8 and row['stats']['Rewarmed'] == 6 and row['stats']['Parked'] == 6

summary = {
    'validWorkloads': len(base['samples'])+len(parking['samples']),
    'parkingMs': {'count':len(park_ms),'min':min(park_ms),'median':statistics.median(park_ms),'max':max(park_ms)},
    'rewarmMs': {'count':len(rewarm_ms),'min':min(rewarm_ms),'median':statistics.median(rewarm_ms),'max':max(rewarm_ms)},
    'dropMs': {'count':len(drop_ms),'min':min(drop_ms),'median':statistics.median(drop_ms),'max':max(drop_ms)},
    'artifactChecks':'exit status, source identity, deterministic row mix, lifecycle errors, resource caps and actual parking/rewarming counts passed',
}
(root/'analysis.json').write_text(json.dumps(summary,indent=2)+'\n')

fig, axes = plt.subplots(2,1,figsize=(11,6.7),sharey=True)
for ax, row in zip(axes,[parking['results'][0],parking['results'][1]]):
    end = row['phases'][-1]['phaseStartMs']+row['phases'][-1]['wallMs']
    points = [p for p in row['timeline'] if p['TimeMs'] <= end]
    x = [p['TimeMs']/1000 for p in points]
    ax.step(x,[p['State']['Capacity'] for p in points],where='post',label='Active databases',color='#0b766e',linewidth=2.2)
    ax.step(x,[p['State']['Reserved'] for p in points],where='post',label='Physical databases reserved',color='#6b7280',linestyle='--',linewidth=1.5)
    ax.step(x,[p['State']['Queued'] for p in points],where='post',label='Queued requests',color='#d97706',linewidth=1.2)
    for phase in row['phases']:
        lo=phase['phaseStartMs']/1000;hi=lo+phase['wallMs']/1000
        ax.axvspan(lo,hi,color='#2563eb',alpha=.06 if phase['workers']==1 else .15)
        label=f"{phase['workers']} worker"+('s' if phase['workers']!=1 else '')
        ax.text((lo+hi)/2,8.4,label,ha='center',fontsize=9,color='#334155')
    idle = row['idleSnapshot']
    lo=idle['startMs']/1000;hi=lo+idle['wallMs']/1000
    ax.text((lo+hi)/2,8.4,'6 seconds idle',ha='center',fontsize=9,color='#334155')
    ax.set_title('Park idle databases' if row['name'].startswith('park') else 'Drop idle databases',loc='left',fontsize=12,weight='bold',pad=15)
    ax.set_ylim(-.2,8.9);ax.set_yticks([0,2,4,6,8]);ax.set_xlim(0,end/1000)
    ax.set_ylabel('Count');ax.grid(axis='y',alpha=.2);ax.spines[['top','right']].set_visible(False)
axes[0].legend(loc='upper left',bbox_to_anchor=(0,-.15),ncol=3,frameon=False,fontsize=9)
axes[1].set_xlabel('Seconds since pool startup (shutdown excluded)')
fig.suptitle('Adaptive pool: first park/drop pair, identical mixed workloads',fontsize=14,weight='bold',y=.98)
fig.tight_layout(h_pad=2.4,rect=(0,0,1,.94))
fig.savefig(root/'capacity-timeline.png',dpi=160)
fig.savefig(root/'capacity-timeline.svg')
print(json.dumps(summary,indent=2))
