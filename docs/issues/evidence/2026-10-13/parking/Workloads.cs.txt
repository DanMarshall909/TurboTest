static class Workloads
{
    public static int Scale(int iteration,int worker)=>((iteration+worker)%10) switch{0=>100,1 or 2 or 3=>10,_=>1};
    public static void Check()
    {
        int[] expected=[100,10,10,10,1,1,1,1,1,1];
        for(var worker=0;worker<8;worker++)
        for(var iteration=0;iteration<30;iteration++)
        {
            var actual=Scale(iteration,worker);var correct=expected[(iteration+worker)%10];
            if(actual!=correct)throw new InvalidDataException($"workload mix: worker={worker}, iteration={iteration}, expected scale={correct}, actual={actual}");
        }
        Console.WriteLine("WORKLOAD CHECKS PASSED: 240 deterministic inputs; 60% small, 30% medium, 10% large");
    }
}
