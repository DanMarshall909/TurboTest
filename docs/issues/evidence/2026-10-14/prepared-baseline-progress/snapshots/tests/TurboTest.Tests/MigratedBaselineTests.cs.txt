using Microsoft.EntityFrameworkCore;
using Microsoft.EntityFrameworkCore.Infrastructure;
using Microsoft.EntityFrameworkCore.Migrations;
using Npgsql;
using TurboTest.EFCore;

namespace TurboTest.Tests;

public class MigratedBaselineTests
{
    [Fact]
    [Trait("Category", "PostgreSQL")]
    public async Task Recreation_preserves_migration_only_schema_history_seed_and_generated_identity()
    {
        var seeds = 0;
        var baseline = new TtBaselineDefinition<MigrationContext>("migration-v1", TtSchemaInitialization.Migrations,
            async (context, cancellation) =>
            {
                seeds++;
                await context.Database.ExecuteSqlRawAsync("""
                    INSERT INTO "MigrationOnly" ("Id", "Label", "Version") VALUES (600, 'migration-seed', 73);
                    SELECT setval(pg_get_serial_sequence('"MigrationOnly"', 'Id'), 600, true);
                    """, cancellation);
            });
        await using var fixture = await Tt.CreateFactoryAsync(baseline, o => o.PoolSize = 1);
        for (var i = 0; i < 2; i++)
        {
            await using var lease = await fixture.CreateLeaseAsync();
            Assert.Equal(1, seeds);
            Assert.Equal(["202610140001_Baseline"], await lease.Context.Database.GetAppliedMigrationsAsync());
            var connection = (NpgsqlConnection)lease.Context.Database.GetDbConnection();
            await connection.OpenAsync();
            await using (var read = new NpgsqlCommand("SELECT \"Id\", \"Label\", \"Version\" FROM \"MigrationOnly\"", connection))
            await using (var reader = await read.ExecuteReaderAsync())
            {
                Assert.True(await reader.ReadAsync());
                Assert.Equal((600, "migration-seed", 73), (reader.GetInt32(0), reader.GetString(1), reader.GetInt32(2)));
                Assert.False(await reader.ReadAsync());
            }
            await using var insert = new NpgsqlCommand("""
                INSERT INTO "MigrationOnly" ("Label", "Version") VALUES ('ordinary', 91) RETURNING "Id"
                """, connection);
            Assert.Equal(601, await insert.ExecuteScalarAsync());
            await lease.Context.Database.ExecuteSqlRawAsync("DROP TABLE \"__EFMigrationsHistory\"");
        }
    }

    public sealed class MigrationContext(DbContextOptions<MigrationContext> options) : DbContext(options);

    [DbContext(typeof(MigrationContext))]
    [Migration("202610140001_Baseline")]
    public sealed class BaselineMigration : Migration
    {
        protected override void Up(MigrationBuilder migrationBuilder) => migrationBuilder.Sql("""
            CREATE TABLE "MigrationOnly" (
              "Id" integer GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
              "Label" text NOT NULL,
              "Version" integer NOT NULL
            )
            """);

        protected override void Down(MigrationBuilder migrationBuilder) =>
            migrationBuilder.Sql("DROP TABLE \"MigrationOnly\"");
    }
}
